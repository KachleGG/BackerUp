// Database Schema
// https://dbml.dbdiagram.io/docs

Table BackupJobs {
  id int [pk, increment]
  method BackupMethod [not null]
  timing varchar [not null]
  created_at timestamp [default: `now()`]
}

Table Sources {
  id int [pk, increment]
  job_id int [not null]
  path varchar [not null]
}

Table Targets {
  id int [pk, increment]
  job_id int [not null]
  path varchar [not null]
}

Table Retention {
  id int [pk, increment]
  job_id int [unique, not null]
  count int [not null, default: 3]
  size int [not null, default: 1]
}

Table Clients {
  id uuid [pk]
  name varchar [not null]
  is_active bool [not null, default: true]
  created_at timestamp [default: `now()`]
}

Table JobsClients {
  id int [pk, increment]
  job_id int [not null]
  client_id uuid [not null]
  is_active bool [not null, default: true]
}

Table Users {
  id uuid [pk]
  username varchar [unique, not null]
  password varchar [not null]
  created_at timestamp [default: `now()`]
}

Table Logs {
  id int [pk, increment]
  jobs_clients_id int [not null]
  level Level [not null]
  description text [not null]
  created_at timestamp [default: `now()`]
}

enum BackupMethod {
  Full
  Incremental
  Differential
}

enum Level {
  Info
  Warning
  Error
}
